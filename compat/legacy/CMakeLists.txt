cmake_minimum_required(VERSION 3.16)
project(mnm_scene_snapshot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets)
if(NOT TARGET mnm-scene-renderer)
    add_subdirectory(../../renderer/scenes "${CMAKE_CURRENT_BINARY_DIR}/scenes")
endif()
add_library(mnm-scene-snapshot STATIC scene_snapshot.cpp world_frame.cpp world_channel.cpp world_resources.cpp)
target_include_directories(mnm-scene-snapshot PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-scene-snapshot PUBLIC mnm-scene-renderer)
target_compile_options(mnm-scene-snapshot PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-scene-snapshot-preview snapshot_main.cpp)
target_link_libraries(mnm-scene-snapshot-preview PRIVATE mnm-scene-snapshot)
target_compile_options(mnm-scene-snapshot-preview PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-frame-preview world_frame_main.cpp)
target_link_libraries(mnm-world-frame-preview PRIVATE mnm-scene-snapshot)
target_compile_options(mnm-world-frame-preview PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-history-preview world_history_main.cpp)
target_link_libraries(mnm-world-history-preview PRIVATE mnm-scene-snapshot)
target_compile_options(mnm-world-history-preview PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-world-live-session STATIC ../../apps/qt-shell/live_world_session.cpp ../../apps/qt-shell/gl_viewport.cpp)
target_include_directories(mnm-world-live-session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/qt-shell")
target_link_libraries(mnm-world-live-session PUBLIC mnm-scene-snapshot Qt6::Widgets Qt6::OpenGLWidgets)
target_compile_options(mnm-world-live-session PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-live ../../apps/qt-shell/world_live_main.cpp)
target_link_libraries(mnm-world-live PRIVATE mnm-world-live-session)
target_compile_options(mnm-world-live PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(world-channel-test ../../tests/world-channel-test.cpp)
    target_link_libraries(world-channel-test PRIVATE mnm-scene-snapshot)
    target_compile_options(world-channel-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-world-channel COMMAND world-channel-test)
    set_tests_properties(native-world-channel PROPERTIES TIMEOUT 30)
    add_executable(scene-snapshot-test ../../tests/scene-snapshot-test.cpp)
    target_link_libraries(scene-snapshot-test PRIVATE mnm-scene-snapshot Qt6::Gui)
    target_compile_options(scene-snapshot-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_executable(world-live-session-test ../../tests/world-live-session-test.cpp)
    target_link_libraries(world-live-session-test PRIVATE mnm-world-live-session)
    target_compile_options(world-live-session-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-world-live-session COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:world-live-session-test>)
    set_tests_properties(native-world-live-session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    add_test(NAME native-scene-snapshot COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:scene-snapshot-test>)
    set_tests_properties(native-scene-snapshot PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    add_executable(world-frame-test ../../tests/world-frame-test.cpp)
    target_link_libraries(world-frame-test PRIVATE mnm-scene-snapshot Qt6::Gui)
    target_compile_options(world-frame-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-world-frame COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:world-frame-test>)
    set_tests_properties(native-world-frame PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
