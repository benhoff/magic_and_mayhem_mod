cmake_minimum_required(VERSION 3.16)
project(mnm_canvas_producers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
if(NOT TARGET mnm-sprite-loader)
 add_subdirectory(../../../assets "${CMAKE_CURRENT_BINARY_DIR}/assets")
endif()
add_library(mnm-canvas-sequence STATIC ../../../renderer/canvas_sequence.cpp ../../../renderer/dib.cpp)
target_link_libraries(mnm-canvas-sequence PUBLIC mnm-sprite-loader Qt6::Gui)
target_compile_options(mnm-canvas-sequence PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-canvas-producers STATIC ../canvas_producers.cpp)
target_link_libraries(mnm-canvas-producers PUBLIC mnm-canvas-sequence mnm-jpeg-loader mnm-pcx-loader)
target_compile_options(mnm-canvas-producers PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-canvas-producers-preview ../canvas_producers_main.cpp)
target_link_libraries(mnm-canvas-producers-preview PRIVATE mnm-canvas-producers Qt6::Core)
target_compile_options(mnm-canvas-producers-preview PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(canvas-sequence-test ../../../tests/canvas-sequence-test.cpp)
 target_link_libraries(canvas-sequence-test PRIVATE mnm-canvas-producers)
 target_compile_options(canvas-sequence-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME native-canvas-sequence COMMAND canvas-sequence-test)
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets)
if(NOT TARGET mnm-renderer)
 add_subdirectory(../../../renderer "${CMAKE_CURRENT_BINARY_DIR}/renderer")
endif()
if(NOT TARGET mnm-scene-renderer)
 add_subdirectory(../../../renderer/scenes "${CMAKE_CURRENT_BINARY_DIR}/scenes")
endif()
add_library(mnm-canvas-world STATIC ../canvas_world.cpp ../world_frame.cpp ../world_resources.cpp ../scene_snapshot.cpp)
target_link_libraries(mnm-canvas-world PUBLIC mnm-canvas-producers mnm-scene-renderer)
target_compile_options(mnm-canvas-world PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
 find_program(XVFB_RUN xvfb-run REQUIRED)
 add_executable(canvas-world-test ../../../tests/canvas-world-test.cpp)
 target_include_directories(canvas-world-test PRIVATE ../../../tests)
 target_link_libraries(canvas-world-test PRIVATE mnm-canvas-world)
 target_compile_options(canvas-world-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME native-canvas-world-handoff COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:canvas-world-test>)
 set_tests_properties(native-canvas-world-handoff PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
add_executable(mnm-canvas-producers-live ../../../apps/qt-shell/canvas_producer_main.cpp ../../../apps/qt-shell/canvas_producer_session.cpp ../../../apps/qt-shell/gl_viewport.cpp)
target_link_libraries(mnm-canvas-producers-live PRIVATE mnm-canvas-world Qt6::Widgets Qt6::OpenGLWidgets)
target_compile_options(mnm-canvas-producers-live PRIVATE -Wall -Wextra -Werror -pedantic)
