cmake_minimum_required(VERSION 3.16)
project(mnm_world_rolling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET mnm-canvas-world)
  add_subdirectory(../canvas-producers "${CMAKE_CURRENT_BINARY_DIR}/canvas-producers")
endif()
add_library(mnm-world-rolling STATIC ../world_rolling_channel.cpp ../world_rolling_packet.cpp ../world_rolling_session.cpp)
target_link_libraries(mnm-world-rolling PUBLIC mnm-canvas-world)
target_compile_options(mnm-world-rolling PRIVATE -Wall -Wextra -Werror -pedantic)
