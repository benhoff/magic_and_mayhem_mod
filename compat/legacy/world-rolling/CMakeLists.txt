cmake_minimum_required(VERSION 3.16)
project(mnm_world_rolling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET mnm-canvas-world)
  add_subdirectory(../canvas-producers "${CMAKE_CURRENT_BINARY_DIR}/canvas-producers")
endif()
add_library(mnm-world-rolling STATIC ../world_rolling_channel.cpp ../world_rolling_packet.cpp ../world_rolling_session.cpp ../world_rolling_handoff.cpp)
target_link_libraries(mnm-world-rolling PUBLIC mnm-canvas-world)
target_compile_options(mnm-world-rolling PRIVATE -Wall -Wextra -Werror -pedantic)

find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets)
add_executable(mnm-world-rolling-live ../../../apps/qt-shell/world_rolling_main.cpp ../../../apps/qt-shell/gl_viewport.cpp)
target_link_libraries(mnm-world-rolling-live PRIVATE mnm-world-rolling Qt6::Widgets Qt6::OpenGLWidgets)
target_compile_options(mnm-world-rolling-live PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
  add_executable(world-rolling-handoff-test ../../../tests/world-rolling-handoff-test.cpp)
  target_link_libraries(world-rolling-handoff-test PRIVATE mnm-world-rolling)
  target_compile_options(world-rolling-handoff-test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
