cmake_minimum_required(VERSION 3.16)
project(mnm_drawing_scene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET mnm-canvas-world)
    add_subdirectory(../canvas-producers "${CMAKE_CURRENT_BINARY_DIR}/producers")
endif()
add_library(mnm-drawing-scene STATIC ../drawing_scene.cpp)
target_link_libraries(mnm-drawing-scene PUBLIC mnm-canvas-world)
target_compile_options(mnm-drawing-scene PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-drawing-scene-replay ../drawing_scene_main.cpp)
target_link_libraries(mnm-drawing-scene-replay PRIVATE mnm-drawing-scene)
target_compile_options(mnm-drawing-scene-replay PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_executable(drawing-scene-test ../../../tests/drawing-scene-test.cpp)
    target_link_libraries(drawing-scene-test PRIVATE mnm-drawing-scene)
    target_compile_options(drawing-scene-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-drawing-scene COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:drawing-scene-test>)
    set_tests_properties(native-drawing-scene PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
