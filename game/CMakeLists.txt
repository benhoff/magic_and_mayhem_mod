cmake_minimum_required(VERSION 3.16)
project(mnm_native_world LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mnm-world STATIC simulation/world.cpp simulation/movement.cpp persistence/snapshot.cpp)
target_include_directories(mnm-world PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(mnm-world PRIVATE -Wall -Wextra -Werror -pedantic)
if(UNIX)
    target_sources(mnm-world PRIVATE persistence/store_posix.cpp)
else()
    message(FATAL_ERROR "Native snapshot file publication currently requires POSIX")
endif()
if(NOT TARGET mnm_pathfinding)
    add_subdirectory(../reconstruction/pathfinding pathfinding)
endif()
add_subdirectory(../reconstruction/motion motion)
add_library(mnm-frozen-navigation STATIC ../apps/world-sandbox/frozen_navigation.cpp)
target_include_directories(mnm-frozen-navigation PUBLIC ../apps/world-sandbox)
target_link_libraries(mnm-frozen-navigation PUBLIC mnm-world PRIVATE mnm_pathfinding mnm_motion)
target_compile_options(mnm-frozen-navigation PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-sandbox ../apps/world-sandbox/main.cpp)
target_link_libraries(mnm-world-sandbox PRIVATE mnm-frozen-navigation)
target_compile_options(mnm-world-sandbox PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(native-movement-test ../tests/native-movement-test.cpp)
    target_link_libraries(native-movement-test PRIVATE mnm-world)
    target_compile_options(native-movement-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-creature-movement COMMAND native-movement-test)
    add_executable(native-world-test ../tests/native-world-test.cpp)
    target_link_libraries(native-world-test PRIVATE mnm-world)
    target_compile_options(native-world-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-world-lifecycle COMMAND native-world-test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME native-fine-motion-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-native-fine-motion.py" "$<TARGET_FILE:mnm-world-sandbox>")
    add_test(NAME native-movement-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-native-movement-session.py" "$<TARGET_FILE:mnm-world-sandbox>")
    add_test(NAME native-world-wire COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-native-world.py" "$<TARGET_FILE:mnm-world-sandbox>")
endif()
