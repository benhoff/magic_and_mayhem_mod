cmake_minimum_required(VERSION 3.16)
project(mnm_native_world LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mnm-world STATIC simulation/world.cpp simulation/movement.cpp persistence/snapshot.cpp)
target_include_directories(mnm-world PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(mnm-world PRIVATE -Wall -Wextra -Werror -pedantic)
if(UNIX)
    target_sources(mnm-world PRIVATE persistence/store_posix.cpp)
else()
    message(FATAL_ERROR "Native snapshot file publication currently requires POSIX")
endif()
if(NOT TARGET mnm_pathfinding)
    add_subdirectory(../reconstruction/pathfinding pathfinding)
endif()
add_subdirectory(../reconstruction/motion motion)
if(NOT TARGET mnm-animation-model)
    add_subdirectory(../reconstruction/animation animation)
endif()
add_library(mnm-sandbox-animation-decoder STATIC ../assets/animation_decode.cpp)
target_include_directories(mnm-sandbox-animation-decoder PUBLIC ../assets)
target_compile_options(mnm-sandbox-animation-decoder PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-frozen-navigation STATIC ../apps/world-sandbox/frozen_navigation.cpp)
target_include_directories(mnm-frozen-navigation PUBLIC ../apps/world-sandbox)
target_link_libraries(mnm-frozen-navigation PUBLIC mnm-world PRIVATE mnm_pathfinding mnm_motion mnm-animation-model mnm-sandbox-animation-decoder)
target_compile_options(mnm-frozen-navigation PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-sandbox ../apps/world-sandbox/main.cpp)
target_link_libraries(mnm-world-sandbox PRIVATE mnm-frozen-navigation)
target_compile_options(mnm-world-sandbox PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
