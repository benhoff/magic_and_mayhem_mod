cmake_minimum_required(VERSION 3.16)
project(mnm_headless_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Gui OpenGL GuiPrivate CorePrivate)
find_package(OpenGL REQUIRED COMPONENTS EGL)
get_filename_component(MNM_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "Build renderer tests separately" FORCE)
add_subdirectory("${MNM_REPO_ROOT}/renderer" "${CMAKE_BINARY_DIR}/renderer" EXCLUDE_FROM_ALL)
add_library(mnm_headless_egl MODULE egl_platform.cpp)
set_target_properties(mnm_headless_egl PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
target_link_libraries(mnm_headless_egl PRIVATE Qt6::GuiPrivate Qt6::CorePrivate OpenGL::EGL)
add_executable(mnm-headless-profile profile.cpp)
target_link_libraries(mnm-headless-profile PRIVATE mnm-renderer Qt6::Gui Qt6::OpenGL)
target_compile_options(mnm-headless-profile PRIVATE -g -fno-omit-frame-pointer)
add_subdirectory("${MNM_REPO_ROOT}/compat/legacy/world-rolling" "${CMAKE_BINARY_DIR}/world-rolling" EXCLUDE_FROM_ALL)
add_executable(mnm-resident-profile resident_profile.cpp)
target_link_libraries(mnm-resident-profile PRIVATE mnm-world-rolling Qt6::Gui Qt6::OpenGL)
target_compile_options(mnm-resident-profile PRIVATE -Wall -Wextra -Werror -pedantic -g -fno-omit-frame-pointer)
add_executable(mnm-gpu-timing-fixture "${MNM_REPO_ROOT}/tests/render-gpu-timing-test.cpp")
target_link_libraries(mnm-gpu-timing-fixture PRIVATE mnm-renderer)
target_compile_options(mnm-gpu-timing-fixture PRIVATE -Wall -Wextra -Werror -pedantic)
find_package(Threads REQUIRED)
add_executable(mnm-resident-capture-fixture "${MNM_REPO_ROOT}/tests/resident-profile-capture-test.cpp" "${MNM_REPO_ROOT}/apps/qt-shell/resident_profile_capture.cpp")
target_link_libraries(mnm-resident-capture-fixture PRIVATE Qt6::Core Threads::Threads)
target_compile_options(mnm-resident-capture-fixture PRIVATE -Wall -Wextra -Werror -pedantic)

add_executable(mnm-palette-row-fixture "${MNM_REPO_ROOT}/tests/palette-row-cache-test.cpp")
target_link_libraries(mnm-palette-row-fixture PRIVATE mnm-renderer)
target_compile_options(mnm-palette-row-fixture PRIVATE -Wall -Wextra -Werror -pedantic)
