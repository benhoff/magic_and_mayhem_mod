cmake_minimum_required(VERSION 3.16)
project(mnm_headless_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Gui OpenGL GuiPrivate CorePrivate)
find_package(OpenGL REQUIRED COMPONENTS EGL)
get_filename_component(MNM_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "Build renderer tests separately" FORCE)
add_subdirectory("${MNM_REPO_ROOT}/renderer" "${CMAKE_BINARY_DIR}/renderer" EXCLUDE_FROM_ALL)
add_library(mnm_headless_egl MODULE egl_platform.cpp)
set_target_properties(mnm_headless_egl PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
target_link_libraries(mnm_headless_egl PRIVATE Qt6::GuiPrivate Qt6::CorePrivate OpenGL::EGL)
add_executable(mnm-headless-profile profile.cpp)
target_link_libraries(mnm-headless-profile PRIVATE mnm-renderer Qt6::Gui Qt6::OpenGL)
target_compile_options(mnm-headless-profile PRIVATE -g -fno-omit-frame-pointer)
