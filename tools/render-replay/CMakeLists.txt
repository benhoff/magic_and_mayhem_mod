cmake_minimum_required(VERSION 3.16)
project(mnm_world_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZSTD REQUIRED libzstd>=1.4)
get_filename_component(MNM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "Build focused controls separately" FORCE)
add_subdirectory("${MNM_ROOT}/renderer" renderer EXCLUDE_FROM_ALL)
add_subdirectory("${MNM_ROOT}/assets" assets EXCLUDE_FROM_ALL)
add_subdirectory("${MNM_ROOT}/compat/legacy/world-rolling" world-rolling EXCLUDE_FROM_ALL)
add_executable(mnm-world-replay main.cpp "${MNM_ROOT}/apps/qt-shell/gl_viewport.cpp")
target_include_directories(mnm-world-replay PRIVATE "${MNM_ROOT}/apps/qt-shell")
target_include_directories(mnm-world-replay SYSTEM PRIVATE ${ZSTD_INCLUDE_DIRS})
target_link_libraries(mnm-world-replay PRIVATE mnm-world-rolling Qt6::Widgets Qt6::OpenGLWidgets)
target_compile_options(mnm-world-replay PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-replay-capture-fixture EXCLUDE_FROM_ALL "${MNM_ROOT}/tests/resident-profile-capture-test.cpp" "${MNM_ROOT}/apps/qt-shell/resident_profile_capture.cpp")
target_include_directories(mnm-replay-capture-fixture SYSTEM PRIVATE ${ZSTD_INCLUDE_DIRS})
target_link_libraries(mnm-replay-capture-fixture PRIVATE Qt6::Core)
target_compile_options(mnm-replay-capture-fixture PRIVATE -UNDEBUG -Wall -Wextra -Werror -pedantic)
