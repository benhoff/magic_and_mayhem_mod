cmake_minimum_required(VERSION 3.16)
project(mnm_font_layout LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Gui)
add_subdirectory(../../assets "${CMAKE_CURRENT_BINARY_DIR}/assets")
add_library(font-layout STATIC ../../compat/legacy/font_layout.cpp ../../compat/legacy/font_text.cpp
  ../../reconstruction/rendering/font_layout.cpp ../../reconstruction/rendering/font_byte.cpp
  ../../renderer/canvas_sequence.cpp ../../renderer/minimap/minimap.cpp ../../renderer/minimap/overlays.cpp)
target_link_libraries(font-layout PUBLIC mnm-sft-loader Qt6::Gui)
target_compile_options(font-layout PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(font-layout-native ../font-layout-native.cpp)
target_link_libraries(font-layout-native PRIVATE font-layout)
target_compile_options(font-layout-native PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(font-layout-test ../font-layout-test.cpp)
target_link_libraries(font-layout-test PRIVATE font-layout)
target_compile_options(font-layout-test PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_test(NAME font-layout-atomic-admission COMMAND font-layout-test)
