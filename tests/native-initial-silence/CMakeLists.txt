cmake_minimum_required(VERSION 3.16)
project(mnm_native_initial_silence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Independent silent startup fixture" FORCE)
find_package(Qt6 6.8 REQUIRED COMPONENTS Gui Multimedia)
add_subdirectory(../../audio audio)
add_executable(initial-voice-test ../native-voice-backend-test.cpp)
target_link_libraries(initial-voice-test PRIVATE mnm-audio-bridge)
add_executable(initial-media-test ../native-media-backend-test.cpp
    ../../apps/qt-shell/native_playback.cpp ../../apps/qt-shell/playback_audio_devices.cpp)
target_link_libraries(initial-media-test PRIVATE Qt6::Gui Qt6::Multimedia)
add_executable(initial-music-test ../native-initial-music-test.cpp)
target_link_libraries(initial-music-test PRIVATE mnm-music-bridge)
foreach(target initial-voice-test initial-media-test initial-music-test)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
endforeach()
