cmake_minimum_required(VERSION 3.16)
project(mnm_word_clipping_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_subdirectory(../../renderer/sprites "${CMAKE_CURRENT_BINARY_DIR}/sprites")
add_executable(word-clipping-gpu ../word-clipping-gpu.cpp)
target_link_libraries(word-clipping-gpu PRIVATE mnm-sprite-renderer)
target_compile_options(word-clipping-gpu PRIVATE -Wall -Wextra -Werror -pedantic)
