cmake_minimum_required(VERSION 3.16)
project(mnm_rendering_thirteenth_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui OpenGL Widgets OpenGLWidgets GuiPrivate CorePrivate)
add_subdirectory(../../apps/world-scene world-scene)
add_subdirectory(../../apps/terrain-preview terrain-scene)
add_subdirectory(../../compat/legacy/canvas-producers canvas)
add_subdirectory(../../compat/legacy/drawing-scene drawing)
find_package(OpenGL REQUIRED COMPONENTS EGL)
add_library(thirteenth_headless_egl MODULE ../../tools/native-render-profile/egl_platform.cpp)
set_target_properties(thirteenth_headless_egl PROPERTIES AUTOMOC ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
target_link_libraries(thirteenth_headless_egl PRIVATE Qt6::GuiPrivate Qt6::CorePrivate OpenGL::EGL)
add_executable(thirteenth-minimap ../rendering-thirteenth-minimap.cpp)
target_link_libraries(thirteenth-minimap PRIVATE mnm-canvas-producers)
target_compile_options(thirteenth-minimap PRIVATE -Wall -Wextra -Werror -pedantic)
