cmake_minimum_required(VERSION 3.16)
project(mnm_font_byte LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Gui)
add_subdirectory(../../assets "${CMAKE_CURRENT_BINARY_DIR}/assets")
add_library(font-text STATIC ../../compat/legacy/font_text.cpp ../../reconstruction/rendering/font_byte.cpp
  ../../renderer/canvas_sequence.cpp ../../renderer/minimap/minimap.cpp ../../renderer/minimap/overlays.cpp)
target_link_libraries(font-text PUBLIC mnm-sft-loader Qt6::Gui)
target_compile_options(font-text PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(font-byte-native ../font-byte-native.cpp)
target_link_libraries(font-byte-native PRIVATE font-text)
target_compile_options(font-byte-native PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(font-byte-test ../font-byte-test.cpp)
target_link_libraries(font-byte-test PRIVATE font-text)
target_compile_options(font-byte-test PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_test(NAME font-byte-atomic-admission COMMAND font-byte-test)
