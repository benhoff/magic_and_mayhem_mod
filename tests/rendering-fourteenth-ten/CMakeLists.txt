cmake_minimum_required(VERSION 3.16)
project(mnm_rendering_fourteenth_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-UNDEBUG)
find_package(Qt6 REQUIRED COMPONENTS Core Gui OpenGL Widgets OpenGLWidgets GuiPrivate CorePrivate)
add_subdirectory(../../compat/legacy/canvas-producers canvas)
add_subdirectory(../../compat/legacy history)
find_package(OpenGL REQUIRED COMPONENTS EGL)
add_library(fourteenth_headless_egl MODULE ../../tools/native-render-profile/egl_platform.cpp)
set_target_properties(fourteenth_headless_egl PROPERTIES AUTOMOC ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
target_link_libraries(fourteenth_headless_egl PRIVATE Qt6::GuiPrivate Qt6::CorePrivate OpenGL::EGL)
