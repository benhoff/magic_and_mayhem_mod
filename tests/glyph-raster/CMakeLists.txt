cmake_minimum_required(VERSION 3.16)
project(mnm_glyph_raster LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Gui)
add_subdirectory(../../assets "${CMAKE_CURRENT_BINARY_DIR}/assets")
add_library(glyph-canvas STATIC ../../renderer/canvas_sequence.cpp ../../renderer/minimap/minimap.cpp ../../renderer/minimap/overlays.cpp)
target_link_libraries(glyph-canvas PUBLIC mnm-sprite-loader Qt6::Gui)
target_compile_options(glyph-canvas PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(glyph-raster-native ../glyph-raster-native.cpp)
target_link_libraries(glyph-raster-native PRIVATE glyph-canvas)
target_compile_options(glyph-raster-native PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(glyph-raster-test ../glyph-raster-test.cpp)
target_link_libraries(glyph-raster-test PRIVATE glyph-canvas)
target_compile_options(glyph-raster-test PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_test(NAME glyph-atomic-admission COMMAND glyph-raster-test)
