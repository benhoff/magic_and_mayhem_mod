cmake_minimum_required(VERSION 3.16)
project(mnm_rendering_low_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui OpenGL Widgets OpenGLWidgets)
add_subdirectory(../native-world-rolling world)
find_package(Threads REQUIRED)
add_executable(low-owner-profile ../rendering-low-owner-profile.cpp)
target_link_libraries(low-owner-profile PRIVATE mnm-world-rolling)
target_compile_options(low-owner-profile PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(low-resident-capture ../resident-profile-capture-test.cpp ../../apps/qt-shell/resident_profile_capture.cpp)
target_link_libraries(low-resident-capture PRIVATE Qt6::Core Threads::Threads)
target_compile_options(low-resident-capture PRIVATE -Wall -Wextra -Werror -pedantic)
