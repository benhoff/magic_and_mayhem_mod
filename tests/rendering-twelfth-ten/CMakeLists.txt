cmake_minimum_required(VERSION 3.16)
project(mnm_rendering_twelfth_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(../rendering-low20 controls)
add_subdirectory(../../apps/drawing-session presenter)
add_executable(twelfth-linear ../rendering-twelfth-linear.cpp)
target_link_libraries(twelfth-linear PRIVATE mnm-canvas-world)
target_compile_options(twelfth-linear PRIVATE -Wall -Wextra -Werror -pedantic)
find_package(Qt6 REQUIRED COMPONENTS Core Gui OpenGL GuiPrivate CorePrivate)
find_package(OpenGL REQUIRED COMPONENTS EGL)
add_library(twelfth_headless_egl MODULE ../../tools/native-render-profile/egl_platform.cpp)
set_target_properties(twelfth_headless_egl PROPERTIES AUTOMOC ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/platforms")
target_link_libraries(twelfth_headless_egl PRIVATE Qt6::GuiPrivate Qt6::CorePrivate OpenGL::EGL)
add_executable(twelfth-resident-profile ../../tools/native-render-profile/resident_profile.cpp)
target_link_libraries(twelfth-resident-profile PRIVATE mnm-world-rolling Qt6::Gui Qt6::OpenGL)
target_compile_options(twelfth-resident-profile PRIVATE -Wall -Wextra -Werror -pedantic)
