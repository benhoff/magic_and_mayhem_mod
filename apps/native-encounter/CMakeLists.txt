cmake_minimum_required(VERSION 3.16)
project(mnm_installed_native_encounter LANGUAGES CXX)
add_subdirectory(../command-session command-session)
# Decode-only entry points have no filesystem/Qt dependency. The unused
# AssetFile wrappers are discarded; no substitute AssetFile implementation.
add_library(mnm-encounter-input-decoders STATIC
  ../../assets/config_loader.cpp ../../assets/packed_container.cpp
  ../../assets/creature_movement.cpp ../../assets/map.cpp
  ../../assets/terrain_catalog.cpp ../world-scene/map_navigation.cpp
  ../../reconstruction/rendering/terrain_map.cpp)
target_compile_features(mnm-encounter-input-decoders PUBLIC cxx_std_17)
target_compile_options(mnm-encounter-input-decoders PRIVATE -ffunction-sections -fdata-sections)
target_include_directories(mnm-encounter-input-decoders PRIVATE ../../assets ../../reconstruction/rendering ../../reconstruction/pathfinding ../../reconstruction/motion)
target_link_libraries(mnm-encounter-input-decoders PUBLIC mnm-native-command-session)
add_library(mnm-native-encounter STATIC profile.cpp combat.cpp ../../reconstruction/combat/health.cpp)
target_compile_features(mnm-native-encounter PUBLIC cxx_std_17)
target_compile_options(mnm-native-encounter PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_options(mnm-native-encounter INTERFACE -Wl,--gc-sections)
target_link_libraries(mnm-native-encounter PUBLIC mnm-encounter-input-decoders)
add_executable(native-encounter main.cpp)
target_link_libraries(native-encounter PRIVATE mnm-native-encounter)
target_compile_options(native-encounter PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(native-encounter-test ../../tests/native-encounter-test.cpp)
target_link_libraries(native-encounter-test PRIVATE mnm-native-encounter)
target_compile_options(native-encounter-test PRIVATE -Wall -Wextra -Werror -pedantic)
set(MNM_ENCOUNTER_INSTALL "" CACHE PATH "Installation for bounded encounter test")
if(MNM_ENCOUNTER_INSTALL)
  enable_testing()
  add_test(NAME native-installed-encounter COMMAND native-encounter-test "${MNM_ENCOUNTER_INSTALL}")
endif()
option(MNM_ENCOUNTER_ORIGINAL_REFERENCE "Build private Linux/i386 original trace oracles" OFF)
add_executable(paired-encounter-helper-test ../../tests/paired-encounter-helper-test.cpp)
target_link_libraries(paired-encounter-helper-test PRIVATE mnm-native-encounter)
target_compile_options(paired-encounter-helper-test PRIVATE -Wall -Wextra -Werror)
foreach(suite combat control motion)
  add_executable(encounter-${suite}-trace-reference ../../tests/encounter-${suite}-trace-reference.cpp)
  target_link_libraries(encounter-${suite}-trace-reference PRIVATE mnm-native-encounter)
  target_compile_options(encounter-${suite}-trace-reference PRIVATE -Wall -Wextra -Werror $<$<NOT:$<BOOL:${MNM_ENCOUNTER_ORIGINAL_REFERENCE}>>:-pedantic>)
  target_include_directories(encounter-${suite}-trace-reference PRIVATE ../../assets ../../reconstruction/combat ../../reconstruction/animation)
endforeach()
if(MNM_ENCOUNTER_ORIGINAL_REFERENCE)
  target_compile_definitions(paired-encounter-helper-test PRIVATE MNM_PAIRED_REFERENCE)
  add_executable(paired-encounter-reference ../../tests/paired-encounter-reference.cpp)
  target_link_libraries(paired-encounter-reference PRIVATE mnm-native-encounter)
  target_compile_options(paired-encounter-reference PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(encounter-combat-trace-reference PRIVATE MNM_HEALTH_REFERENCE)
  target_compile_definitions(encounter-control-trace-reference PRIVATE MNM_CONTROL_REFERENCE)
  target_compile_definitions(encounter-motion-trace-reference PRIVATE MNM_EXECUTION_REFERENCE)
  target_link_libraries(encounter-control-trace-reference PRIVATE mnm-native-command-session)
endif()

target_link_libraries(paired-encounter-native PRIVATE mnm-native-encounter)
target_compile_options(paired-encounter-native PRIVATE -Wall -Wextra -Werror -pedantic)
