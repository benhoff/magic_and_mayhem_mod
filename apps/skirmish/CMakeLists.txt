cmake_minimum_required(VERSION 3.16)
project(mnm_native_skirmish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
include(CTest)
add_subdirectory(../../game world)
add_subdirectory(../../reconstruction/combat original-health)
add_library(mnm-arena STATIC arena.cpp trace.cpp health_rules.cpp)
target_include_directories(mnm-arena PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-arena PUBLIC mnm-skirmish-core PRIVATE mnm-frozen-navigation mnm-original-health)
target_compile_options(mnm-arena PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-campaign-arena STATIC campaign.cpp)
add_subdirectory(../../assets asset-inputs)
add_subdirectory(../../reconstruction/rendering queue)
add_library(mnm-campaign-map STATIC map_campaign.cpp ../world-scene/map_navigation.cpp ../../reconstruction/motion/creature_profile.cpp)
target_include_directories(mnm-campaign-map PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../world-scene" "${CMAKE_CURRENT_SOURCE_DIR}/../../reconstruction/motion")
target_link_libraries(mnm-campaign-map PUBLIC mnm-campaign-arena mnm-map-loader mnm-terrain-catalog mnm-sprite-loader PRIVATE mnm-frozen-navigation mnm_pathfinding mnm-sprite-queue mnm-sandbox-animation-decoder Qt6::Core)
target_compile_options(mnm-campaign-map PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-map-campaign-cli campaign_cli.cpp)
target_compile_definitions(mnm-map-campaign-cli PRIVATE MNM_MAP_CAMPAIGN_CLI)
target_link_libraries(mnm-map-campaign-cli PRIVATE mnm-campaign-map)
add_executable(map-campaign-test ../../tests/map-campaign-test.cpp)
target_link_libraries(map-campaign-test PRIVATE mnm-campaign-map)
target_compile_options(map-campaign-test PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(mnm-campaign-arena PUBLIC mnm-arena mnm-campaign-session)
target_compile_options(mnm-campaign-arena PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-campaign-cli campaign_cli.cpp)
target_link_libraries(mnm-campaign-cli PRIVATE mnm-campaign-arena)
target_compile_options(mnm-campaign-cli PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-skirmish-window STATIC controller.cpp window.cpp ../world-scene/playback.cpp)
target_include_directories(mnm-skirmish-window PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../world-scene")
target_compile_definitions(mnm-skirmish-window PUBLIC QT_NO_KEYWORDS)
target_link_libraries(mnm-skirmish-window PUBLIC mnm-campaign-arena Qt6::Widgets)
target_compile_options(mnm-skirmish-window PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-skirmish main.cpp)
target_link_libraries(mnm-skirmish PRIVATE mnm-skirmish-window)
target_compile_options(mnm-skirmish PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-skirmish-cli cli.cpp)
target_link_libraries(mnm-skirmish-cli PRIVATE mnm-arena)
target_compile_options(mnm-skirmish-cli PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(campaign-arena-test ../../tests/campaign-arena-test.cpp)
    target_link_libraries(campaign-arena-test PRIVATE mnm-campaign-arena)
    target_compile_options(campaign-arena-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-campaign-arena-contract COMMAND campaign-arena-test)
    add_test(NAME native-campaign-arena-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test-campaign-arena.py" "$<TARGET_FILE:mnm-campaign-cli>" "$<TARGET_FILE:campaign-arena-test>")
    add_executable(campaign-window-test main.cpp ../../tests/campaign-window-driver.cpp)
    target_compile_definitions(campaign-window-test PRIVATE MNM_CAMPAIGN_WINDOW_TEST)
    target_link_libraries(campaign-window-test PRIVATE mnm-skirmish-window)
    target_compile_options(campaign-window-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-campaign-window COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test-campaign-window.py" "$<TARGET_FILE:campaign-window-test>" "$<TARGET_FILE:mnm-campaign-cli>" "$<TARGET_FILE:campaign-arena-test>")
    set_tests_properties(native-campaign-window PROPERTIES TIMEOUT 180)

    add_executable(skirmish-test ../../tests/skirmish-test.cpp)

    target_compile_options(skirmish-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-skirmish-contract COMMAND skirmish-test)

    add_test(NAME native-skirmish-process COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test-skirmish.py" "$<TARGET_FILE:mnm-skirmish-cli>")
    add_test(NAME native-skirmish-window COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test-skirmish-window.py" "$<TARGET_FILE:skirmish-window-test>" "$<TARGET_FILE:mnm-skirmish-cli>")
    set_tests_properties(native-skirmish-window PROPERTIES TIMEOUT 180)
