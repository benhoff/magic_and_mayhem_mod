cmake_minimum_required(VERSION 3.16)
project(mnm_world_scene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
include(CTest)
add_subdirectory(../../game world)
add_subdirectory(../../renderer/scenes "${CMAKE_CURRENT_BINARY_DIR}/scenes")
add_subdirectory(../../reconstruction/rendering queue)
add_library(mnm-map-navigation STATIC map_navigation.cpp ../../reconstruction/motion/creature_profile.cpp)
target_include_directories(mnm-map-navigation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../reconstruction/motion")
target_link_libraries(mnm-map-navigation PUBLIC mnm-map-loader mnm-terrain-catalog PRIVATE mnm_pathfinding mnm-sprite-queue mnm-sandbox-animation-decoder)
target_compile_options(mnm-map-navigation PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-map-navigation-export map_navigation_main.cpp)
target_link_libraries(mnm-map-navigation-export PRIVATE mnm-map-navigation mnm-frozen-navigation Qt6::Core)
target_compile_options(mnm-map-navigation-export PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-world-scene STATIC scene.cpp picking.cpp)
target_include_directories(mnm-world-scene PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(mnm-world-scene PUBLIC QT_NO_KEYWORDS)
target_link_libraries(mnm-world-scene PUBLIC mnm-frozen-navigation mnm-scene-renderer mnm-animation-model mnm-sprite-queue mnm-terrain-catalog)
target_compile_options(mnm-world-scene PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-world-controls STATIC orders.cpp movement_controls.cpp scene_canvas.cpp playback.cpp playback_controls.cpp)
target_include_directories(mnm-world-controls PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(mnm-world-controls PUBLIC QT_NO_KEYWORDS)
target_link_libraries(mnm-world-controls PUBLIC mnm-world Qt6::Widgets)
target_compile_options(mnm-world-controls PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-scene-preview main.cpp)
target_link_libraries(mnm-world-scene-preview PRIVATE mnm-world-scene mnm-world-controls mnm-map-navigation Qt6::Widgets)
target_compile_options(mnm-world-scene-preview PRIVATE -Wall -Wextra -Werror -pedantic)
