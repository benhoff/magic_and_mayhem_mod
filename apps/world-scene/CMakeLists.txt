cmake_minimum_required(VERSION 3.16)
project(mnm_world_scene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
include(CTest)
add_subdirectory(../../game world)
add_subdirectory(../../renderer/sprites sprites)
add_subdirectory(../../reconstruction/rendering queue)
add_library(mnm-world-scene STATIC scene.cpp)
target_include_directories(mnm-world-scene PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(mnm-world-scene PUBLIC QT_NO_KEYWORDS)
target_link_libraries(mnm-world-scene PUBLIC mnm-frozen-navigation mnm-sprite-renderer mnm-animation-model mnm-sprite-queue mnm-terrain-catalog)
target_compile_options(mnm-world-scene PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-scene-preview main.cpp)
target_link_libraries(mnm-world-scene-preview PRIVATE mnm-world-scene Qt6::Widgets)
target_compile_options(mnm-world-scene-preview PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
    add_executable(world-scene-test ../../tests/world-scene-test.cpp)
    target_link_libraries(world-scene-test PRIVATE mnm-world-scene)
    target_compile_options(world-scene-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_test(NAME native-world-scene COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:world-scene-test>)
    set_tests_properties(native-world-scene PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
