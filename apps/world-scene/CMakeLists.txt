cmake_minimum_required(VERSION 3.16)
project(mnm_world_scene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
include(CTest)
add_subdirectory(../../game world)
add_subdirectory(../../renderer/sprites sprites)
add_subdirectory(../../reconstruction/rendering queue)
add_library(mnm-map-navigation STATIC map_navigation.cpp)
target_include_directories(mnm-map-navigation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-map-navigation PUBLIC mnm-map-loader mnm-terrain-catalog PRIVATE mnm_pathfinding mnm-sprite-queue)
target_compile_options(mnm-map-navigation PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-map-navigation-export map_navigation_main.cpp)
target_link_libraries(mnm-map-navigation-export PRIVATE mnm-map-navigation Qt6::Core)
target_compile_options(mnm-map-navigation-export PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-world-scene STATIC scene.cpp)
target_include_directories(mnm-world-scene PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(mnm-world-scene PUBLIC QT_NO_KEYWORDS)
target_link_libraries(mnm-world-scene PUBLIC mnm-frozen-navigation mnm-sprite-renderer mnm-animation-model mnm-sprite-queue mnm-terrain-catalog)
target_compile_options(mnm-world-scene PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-world-scene-preview main.cpp)
target_link_libraries(mnm-world-scene-preview PRIVATE mnm-world-scene mnm-map-navigation Qt6::Widgets)
target_compile_options(mnm-world-scene-preview PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
    add_executable(map-navigation-test ../../tests/map-navigation-test.cpp)
    target_link_libraries(map-navigation-test PRIVATE mnm-map-navigation mnm_pathfinding)
    target_compile_options(map-navigation-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-map-navigation COMMAND map-navigation-test)
    add_executable(world-scene-test ../../tests/world-scene-test.cpp)
    target_link_libraries(world-scene-test PRIVATE mnm-world-scene)
    target_compile_options(world-scene-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_test(NAME native-world-scene COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:world-scene-test>)
    set_tests_properties(native-world-scene PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
