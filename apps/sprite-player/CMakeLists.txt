cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_player LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
add_subdirectory(../sprite-timeline "${CMAKE_CURRENT_BINARY_DIR}/timeline")
add_library(mnm-sprite-player STATIC sprite_player.cpp)
target_link_libraries(mnm-sprite-player PUBLIC mnm-sprite-timeline)
target_include_directories(mnm-sprite-player PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(mnm-sprite-window STATIC sprite_window.cpp)
target_link_libraries(mnm-sprite-window PUBLIC mnm-sprite-player Qt6::Widgets)
add_executable(mnm-sprite-player-app main.cpp)
set_target_properties(mnm-sprite-player-app PROPERTIES OUTPUT_NAME mnm-sprite-player)
target_link_libraries(mnm-sprite-player-app PRIVATE mnm-sprite-window)
foreach(target mnm-sprite-player mnm-sprite-window mnm-sprite-player-app)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(sprite-player-test ../../tests/sprite-player-test.cpp)
 target_link_libraries(sprite-player-test PRIVATE mnm-sprite-window Qt6::Test)
 target_compile_options(sprite-player-test PRIVATE -Wall -Wextra -Werror -pedantic)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME sprite-table-application COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/run-surface-wrapper-control.py"
   --log-root "${CMAKE_CURRENT_BINARY_DIR}/player-displays" -- $<TARGET_FILE:sprite-player-test>)
 set_tests_properties(sprite-table-application PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;LP_NUM_THREADS=2" TIMEOUT 180)
endif()
