cmake_minimum_required(VERSION 3.16)
project(mnm_cure_session LANGUAGES CXX)
add_library(mnm-cure-session STATIC session.cpp ../../reconstruction/casting/cure_effect.cpp ../../reconstruction/casting/cure_services.cpp ../../reconstruction/casting/admission.cpp ../../reconstruction/combat/health.cpp ../../reconstruction/audio/voice_contract.cpp ../../reconstruction/audio/voice_lifetime.cpp)
target_include_directories(mnm-cure-session PUBLIC ../../audio)
target_compile_features(mnm-cure-session PUBLIC cxx_std_17)
target_compile_options(mnm-cure-session PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(cure-lab main.cpp)
target_link_libraries(cure-lab PRIVATE mnm-cure-session)
target_compile_options(cure-lab PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(cure-session-test ../../tests/cure-session-test.cpp)
  target_link_libraries(cure-session-test PRIVATE mnm-cure-session)
  target_compile_options(cure-session-test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME cure-session COMMAND cure-session-test)
endif()
