cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_timeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET mnm-sprite-session)
 add_subdirectory(../sprite-session "${CMAKE_CURRENT_BINARY_DIR}/session")
endif()
add_library(mnm-sprite-timeline STATIC sprite_timeline.cpp)
target_include_directories(mnm-sprite-timeline PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-sprite-timeline PUBLIC mnm-sprite-session)
target_compile_options(mnm-sprite-timeline PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(sprite-timeline-test ../../tests/sprite-timeline-test.cpp)
 target_link_libraries(sprite-timeline-test PRIVATE mnm-sprite-timeline)
 target_compile_options(sprite-timeline-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME native-sprite-timeline COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/run-surface-wrapper-control.py" --log-root "${CMAKE_CURRENT_BINARY_DIR}/timeline-displays" -- $<TARGET_FILE:sprite-timeline-test>)
 set_tests_properties(native-sprite-timeline PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;LP_NUM_THREADS=2" TIMEOUT 150)
endif()
