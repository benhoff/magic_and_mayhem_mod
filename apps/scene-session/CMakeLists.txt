cmake_minimum_required(VERSION 3.16)
project(mnm_scene_session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
find_package(Threads REQUIRED)
if(NOT TARGET mnm-scene-renderer)
    add_subdirectory(../../renderer/scenes "${CMAKE_CURRENT_BINARY_DIR}/scenes")
endif()
if(NOT TARGET mnm-sprite-queue)
    add_subdirectory(../../reconstruction/rendering "${CMAKE_CURRENT_BINARY_DIR}/queue")
endif()
add_library(mnm-scene-palette STATIC palette_binding.cpp palette_profile.cpp)
target_include_directories(mnm-scene-palette PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-scene-palette PUBLIC mnm-scene-renderer mnm-profile-loader mnm-sprite-queue Threads::Threads)
target_compile_definitions(mnm-scene-palette PUBLIC QT_NO_KEYWORDS)
target_compile_options(mnm-scene-palette PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS GuiPrivate CorePrivate)
    find_package(OpenGL REQUIRED COMPONENTS EGL)
    add_library(mnm_scene_palette_headless MODULE ../../tools/native-render-profile/egl_platform.cpp)
    set_target_properties(mnm_scene_palette_headless PROPERTIES AUTOMOC ON
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/platforms")
    target_link_libraries(mnm_scene_palette_headless PRIVATE Qt6::GuiPrivate Qt6::CorePrivate OpenGL::EGL)
    add_executable(scene-palette-test ../../tests/scene-palette-test.cpp)
    target_link_libraries(scene-palette-test PRIVATE mnm-scene-palette Qt6::Gui)
    target_compile_options(scene-palette-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_dependencies(scene-palette-test mnm_scene_palette_headless)
    add_test(NAME native-scene-palette-effects COMMAND "${CMAKE_COMMAND}" -E env
        QT_QPA_PLATFORM=mnm-headless-egl
        QT_QPA_PLATFORM_PLUGIN_PATH=$<TARGET_FILE_DIR:mnm_scene_palette_headless>
        MNM_PROFILE_EGL_VENDOR=Mesa LIBGL_ALWAYS_SOFTWARE=1 QT_FORCE_STDERR_LOGGING=1
        $<TARGET_FILE:scene-palette-test>)
    set_tests_properties(native-scene-palette-effects PROPERTIES TIMEOUT 60)
endif()
