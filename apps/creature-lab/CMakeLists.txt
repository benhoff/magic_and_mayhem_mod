cmake_minimum_required(VERSION 3.16)
project(mnm_creature_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
add_subdirectory(../melee-lab melee)
add_library(mnm-creature-profile STATIC profile.cpp)
target_include_directories(mnm-creature-profile PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-creature-profile PUBLIC mnm-creature-encounter PRIVATE mnm-melee-profile mnm-creature-control mnm-original-melee mnm-original-health mnm-animation-model mnm-persistence-loader mnm-animation-loader Qt6::Core)
target_compile_options(mnm-creature-profile PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-creature-window STATIC controller.cpp window.cpp)
target_compile_definitions(mnm-creature-window PUBLIC QT_NO_KEYWORDS)
target_link_libraries(mnm-creature-window PUBLIC mnm-creature-profile Qt6::Widgets)
target_compile_options(mnm-creature-window PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-creature-lab main.cpp)
target_link_libraries(mnm-creature-lab PRIVATE mnm-creature-window)
target_compile_options(mnm-creature-lab PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-creature-cli cli.cpp)
target_link_libraries(mnm-creature-cli PRIVATE mnm-creature-profile)
target_compile_options(mnm-creature-cli PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
 add_executable(native-creature-encounter-test ../../tests/native-creature-encounter-test.cpp)
 target_link_libraries(native-creature-encounter-test PRIVATE mnm-creature-profile)
 target_compile_options(native-creature-encounter-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_executable(creature-window-test main.cpp ../../tests/creature-window-driver.cpp)
 target_compile_definitions(creature-window-test PRIVATE MNM_CREATURE_WINDOW_TEST)
 target_link_libraries(creature-window-test PRIVATE mnm-creature-window)
 target_compile_options(creature-window-test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
