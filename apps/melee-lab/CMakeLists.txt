cmake_minimum_required(VERSION 3.16)
project(mnm_melee_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
include(CTest)
add_subdirectory(../../game world)
add_subdirectory(../../assets assets)
add_subdirectory(../../reconstruction/combat original-combat)
add_library(mnm-melee-profile STATIC profile.cpp ../skirmish/health_rules.cpp)
target_include_directories(mnm-melee-profile PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-melee-profile PUBLIC mnm-melee-exchange PRIVATE mnm-original-melee mnm-original-health mnm-animation-model mnm-persistence-loader mnm-animation-loader Qt6::Core)
target_compile_options(mnm-melee-profile PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-melee-cli cli.cpp)
target_link_libraries(mnm-melee-cli PRIVATE mnm-melee-profile)
target_compile_options(mnm-melee-cli PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-melee-window STATIC controller.cpp window.cpp)
target_link_libraries(mnm-melee-window PUBLIC mnm-melee-profile Qt6::Widgets)
target_compile_options(mnm-melee-window PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-melee-lab main.cpp)
target_link_libraries(mnm-melee-lab PRIVATE mnm-melee-window)
target_compile_options(mnm-melee-lab PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
    add_executable(native-melee-exchange-test ../../tests/native-melee-exchange-test.cpp)
    target_link_libraries(native-melee-exchange-test PRIVATE mnm-melee-profile)
    target_compile_options(native-melee-exchange-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_executable(melee-window-test main.cpp ../../tests/melee-window-driver.cpp)
    target_compile_definitions(melee-window-test PRIVATE MNM_MELEE_WINDOW_TEST)
    target_link_libraries(melee-window-test PRIVATE mnm-melee-window)
    target_compile_options(melee-window-test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
