cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET mnm-sprite-renderer)
 add_subdirectory(../../renderer/sprites "${CMAKE_CURRENT_BINARY_DIR}/sprites")
endif()
add_library(mnm-sprite-batch STATIC ../../renderer/sprites/sprite_batch.cpp)
target_link_libraries(mnm-sprite-batch PUBLIC mnm-sprite-renderer)
target_compile_options(mnm-sprite-batch PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-sprite-session STATIC sprite_session.cpp)
target_include_directories(mnm-sprite-session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-sprite-session PUBLIC mnm-sprite-batch mnm-tag-loader)
target_compile_options(mnm-sprite-session PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(sprite-session-test ../../tests/sprite-session-test.cpp)
 target_link_libraries(sprite-session-test PRIVATE mnm-sprite-session)
 target_compile_options(sprite-session-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME native-sprite-session COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/run-surface-wrapper-control.py" --log-root "${CMAKE_CURRENT_BINARY_DIR}/sprite-session-displays" -- $<TARGET_FILE:sprite-session-test>)
 set_tests_properties(native-sprite-session PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;LP_NUM_THREADS=2" TIMEOUT 150)
endif()
