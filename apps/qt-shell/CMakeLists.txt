cmake_minimum_required(VERSION 3.16)
project(mnm_qt_shell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
add_executable(mnm-qt-shell main.cpp window_host.cpp gl_viewport.cpp frame_stream.cpp)
target_link_libraries(mnm-qt-shell PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL PkgConfig::XCB)
target_compile_options(mnm-qt-shell PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME qt-shell-help COMMAND mnm-qt-shell --help)
    add_executable(frame-stream-test ../../tests/frame-stream-test.cpp frame_stream.cpp)
    target_link_libraries(frame-stream-test PRIVATE Qt6::Gui)
    target_compile_options(frame-stream-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME frame-stream COMMAND frame-stream-test)
    add_executable(render-pixels-test ../../tests/render-pixels-test.cpp)
    target_compile_options(render-pixels-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME render-pixels COMMAND render-pixels-test)
    add_test(NAME qt-shell-startup COMMAND mnm-qt-shell --renderer native --smoke-test)
    set_tests_properties(qt-shell-startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME qt-shell-embedding COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:mnm-qt-shell> --embedding-test)
        add_test(NAME qt-shell-opengl COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:mnm-qt-shell> --opengl-test)
        set_tests_properties(qt-shell-opengl PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 15)
        set_tests_properties(qt-shell-embedding PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb" TIMEOUT 15)
    endif()
endif()
