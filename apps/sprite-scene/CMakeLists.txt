cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_scene LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
include(CTest)
add_subdirectory(../../renderer/sprites "${CMAKE_CURRENT_BINARY_DIR}/sprites")
add_subdirectory(../../reconstruction/animation "${CMAKE_CURRENT_BINARY_DIR}/animation")
add_subdirectory(../../reconstruction/rendering "${CMAKE_CURRENT_BINARY_DIR}/queue")
add_library(mnm-sprite-scene STATIC scene.cpp attachment_recipe.cpp)
target_include_directories(mnm-sprite-scene PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-sprite-scene PUBLIC mnm-sprite-queue mnm-sprite-renderer mnm-animation-model mnm-persistence-loader)
target_compile_options(mnm-sprite-scene PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-sprite-scene-preview main.cpp)
target_link_libraries(mnm-sprite-scene-preview PRIVATE mnm-sprite-scene mnm-animation-loader Qt6::Widgets)
target_compile_options(mnm-sprite-scene-preview PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
    add_executable(attachment-recipe-test ../../tests/attachment-recipe-test.cpp)
    target_link_libraries(attachment-recipe-test PRIVATE mnm-sprite-scene)
    target_compile_options(attachment-recipe-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME animation-attachment-recipe COMMAND attachment-recipe-test)
    add_executable(sprite-scene-test ../../tests/sprite-scene-test.cpp)
    target_link_libraries(sprite-scene-test PRIVATE mnm-sprite-scene)
    target_compile_options(sprite-scene-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_executable(sprite-scene-queue-test ../../tests/sprite-scene-queue-test.cpp)
    target_link_libraries(sprite-scene-queue-test PRIVATE mnm-sprite-scene)
    target_compile_options(sprite-scene-queue-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME opengl-sprite-queue-scene COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:sprite-scene-queue-test>)
        set_tests_properties(opengl-sprite-queue-scene PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
        add_test(NAME opengl-animation-scene COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:sprite-scene-test>)
        set_tests_properties(opengl-animation-scene PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    endif()
endif()
