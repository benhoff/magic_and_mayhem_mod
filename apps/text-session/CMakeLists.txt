cmake_minimum_required(VERSION 3.16)
project(mnm_text_session LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET mnm-sprite-renderer)
 add_subdirectory(../../renderer/sprites "${CMAKE_CURRENT_BINARY_DIR}/sprites")
endif()
add_library(mnm-font-pixels STATIC ../../renderer/fonts/font_pixels.cpp)
target_include_directories(mnm-font-pixels PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../renderer/fonts")
target_link_libraries(mnm-font-pixels PUBLIC mnm-sprite-renderer)
target_compile_options(mnm-font-pixels PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-text-session STATIC text_session.cpp ../../reconstruction/rendering/font_byte.cpp ../../reconstruction/rendering/font_layout.cpp)
target_include_directories(mnm-text-session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-text-session PUBLIC mnm-font-pixels mnm-sft-loader)
target_compile_options(mnm-text-session PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(text-session-test ../../tests/text-session-test.cpp)
 target_link_libraries(text-session-test PRIVATE mnm-text-session)
 target_compile_options(text-session-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME native-text-session COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/run-surface-wrapper-control.py" --log-root "${CMAKE_CURRENT_BINARY_DIR}/text-displays" -- $<TARGET_FILE:text-session-test>)
 set_tests_properties(native-text-session PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;LP_NUM_THREADS=2" TIMEOUT 150)
endif()
