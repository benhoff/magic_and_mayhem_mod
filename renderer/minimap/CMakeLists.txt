cmake_minimum_required(VERSION 3.16)
project(mnm_minimap LANGUAGES CXX)
add_library(mnm-minimap STATIC minimap.cpp)
target_compile_features(mnm-minimap PUBLIC cxx_std_17)
target_compile_options(mnm-minimap PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(minimap-native ../../tests/minimap-native.cpp)
  target_link_libraries(minimap-native PRIVATE mnm-minimap)
  target_compile_options(minimap-native PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME native-minimap-refusals COMMAND minimap-native --selftest)
endif()
