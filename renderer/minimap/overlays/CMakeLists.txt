cmake_minimum_required(VERSION 3.16)
project(mnm_minimap_overlays LANGUAGES CXX)
add_library(mnm-minimap-overlays STATIC ../overlays.cpp)
target_compile_features(mnm-minimap-overlays PUBLIC cxx_std_17)
target_compile_options(mnm-minimap-overlays PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(minimap-overlays-native ../../../tests/minimap-overlays-native.cpp)
  target_link_libraries(minimap-overlays-native PRIVATE mnm-minimap-overlays)
  target_compile_options(minimap-overlays-native PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME native-minimap-overlay-refusals COMMAND minimap-overlays-native --selftest)
endif()
