cmake_minimum_required(VERSION 3.16)
project(mnm_resource_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
if(NOT TARGET mnm-resources)
    add_subdirectory(../../assets/resources "${CMAKE_CURRENT_BINARY_DIR}/resources")
endif()
if(NOT TARGET mnm-sprite-renderer)
    add_subdirectory(../sprites "${CMAKE_CURRENT_BINARY_DIR}/sprites")
endif()
add_library(mnm-resource-renderer STATIC resource_cache.cpp sprite_atlas.cpp)
target_include_directories(mnm-resource-renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-resource-renderer PUBLIC mnm-resources mnm-sprite-renderer)
target_compile_options(mnm-resource-renderer PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-resource-preview preview_main.cpp)
target_link_libraries(mnm-resource-preview PRIVATE mnm-resource-renderer)
target_compile_options(mnm-resource-preview PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(resource-cache-test ../../tests/resource-cache-test.cpp)
    target_link_libraries(resource-cache-test PRIVATE mnm-resource-renderer Qt6::Core)
    target_compile_options(resource-cache-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME native-resource-cache COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:resource-cache-test>)
    set_tests_properties(native-resource-cache PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    add_test(NAME native-resource-preview COMMAND "${XVFB_RUN}" -a "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test-resource-preview.py" $<TARGET_FILE:mnm-resource-preview>)
    set_tests_properties(native-resource-preview PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
