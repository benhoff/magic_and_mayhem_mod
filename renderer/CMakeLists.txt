cmake_minimum_required(VERSION 3.16)
project(mnm_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.5 REQUIRED COMPONENTS Gui OpenGL)
add_library(mnm-renderer STATIC blit.cpp capture.cpp)
target_include_directories(mnm-renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-renderer PUBLIC Qt6::Gui Qt6::OpenGL)
target_compile_options(mnm-renderer PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-render-replay replay_main.cpp)
target_link_libraries(mnm-render-replay PRIVATE mnm-renderer)
target_compile_options(mnm-render-replay PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME renderer-help COMMAND mnm-render-replay --help)
    add_executable(render-blit-test ../tests/render-blit-test.cpp)
    target_link_libraries(render-blit-test PRIVATE mnm-renderer)
    target_compile_options(render-blit-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_executable(render-surfaces-test ../tests/render-surfaces-test.cpp)
    target_link_libraries(render-surfaces-test PRIVATE mnm-renderer)
    target_compile_options(render-surfaces-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME native-capture-reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-render-capture.py")
    if(XVFB_RUN)
        add_test(NAME opengl-native-blits COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:render-blit-test>)
        add_test(NAME opengl-persistent-surfaces COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:render-surfaces-test>)
        add_test(NAME opengl-capture-replay COMMAND "${XVFB_RUN}" -a "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-opengl-replay.py" $<TARGET_FILE:mnm-render-replay>)
        set_tests_properties(opengl-native-blits opengl-capture-replay opengl-persistent-surfaces PROPERTIES
                             ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    endif()
endif()
