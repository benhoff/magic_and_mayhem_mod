cmake_minimum_required(VERSION 3.16)
project(mnm_alternate_canvas LANGUAGES CXX)
add_library(mnm-alternate-canvas INTERFACE)
target_compile_features(mnm-alternate-canvas INTERFACE cxx_std_17)
target_include_directories(mnm-alternate-canvas INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/..")
include(CTest)
if(BUILD_TESTING)
  add_executable(alternate-canvas-owner-test ../../tests/alternate-canvas-owner-test.cpp)
  target_link_libraries(alternate-canvas-owner-test PRIVATE mnm-alternate-canvas)
  target_compile_options(alternate-canvas-owner-test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME native-alternate-canvas-owner COMMAND alternate-canvas-owner-test)
endif()
