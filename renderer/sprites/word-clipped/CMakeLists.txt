cmake_minimum_required(VERSION 3.16)
project(mnm_word_clipped LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(mnm-word-clipped STATIC ../word_clipped.c)
target_include_directories(mnm-word-clipped PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(mnm-word-clipped PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(word-clipped-test ../../../tests/word-clipped-test.c)
    target_link_libraries(word-clipped-test PRIVATE mnm-word-clipped)
    target_compile_options(word-clipped-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-word-clipped COMMAND word-clipped-test)
endif()
