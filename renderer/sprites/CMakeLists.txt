cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET mnm-sprite-loader)
    add_subdirectory(../../assets "${CMAKE_CURRENT_BINARY_DIR}/assets")
endif()
if(NOT TARGET mnm-renderer)
    add_subdirectory(.. "${CMAKE_CURRENT_BINARY_DIR}/renderer")
endif()
add_library(mnm-sprite-renderer STATIC sprite.cpp)
target_include_directories(mnm-sprite-renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-sprite-renderer PUBLIC mnm-renderer mnm-sprite-loader)
target_compile_options(mnm-sprite-renderer PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-sprite-preview preview_main.cpp)
target_link_libraries(mnm-sprite-preview PRIVATE mnm-sprite-renderer)
target_compile_options(mnm-sprite-preview PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(sprite-render-test ../../tests/sprite-render-test.cpp)
    target_link_libraries(sprite-render-test PRIVATE mnm-sprite-renderer)
    target_compile_options(sprite-render-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME opengl-sprite-frames COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:sprite-render-test>)
        set_tests_properties(opengl-sprite-frames PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    endif()
endif()
