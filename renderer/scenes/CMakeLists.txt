cmake_minimum_required(VERSION 3.16)
project(mnm_scene_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
if(NOT TARGET mnm-resource-renderer)
    add_subdirectory(../resources "${CMAKE_CURRENT_BINARY_DIR}/resources")
endif()
add_library(mnm-scene-renderer STATIC scene_renderer.cpp scene_history.cpp)
target_include_directories(mnm-scene-renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-scene-renderer PUBLIC mnm-resource-renderer)
target_compile_options(mnm-scene-renderer PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(sprite-atlas-test ../../tests/sprite-atlas-test.cpp)
    target_link_libraries(sprite-atlas-test PRIVATE mnm-scene-renderer)
    target_compile_options(sprite-atlas-test PRIVATE -Wall -Wextra -Werror -pedantic)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_test(NAME opengl-sprite-atlas COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:sprite-atlas-test>)
    set_tests_properties(opengl-sprite-atlas PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)

    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_executable(scene-upload-test ../../tests/scene-upload-test.cpp)
    target_link_libraries(scene-upload-test PRIVATE mnm-scene-renderer)
    target_compile_options(scene-upload-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME opengl-scene-upload-budget COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:scene-upload-test>)
    set_tests_properties(opengl-scene-upload-budget PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    add_executable(scene-renderer-test ../../tests/scene-renderer-test.cpp)
    target_link_libraries(scene-renderer-test PRIVATE mnm-scene-renderer Qt6::Gui)
    target_compile_options(scene-renderer-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-shared-scene-renderer COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:scene-renderer-test>)
    set_tests_properties(native-shared-scene-renderer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    add_executable(scene-history-test ../../tests/scene-history-test.cpp)
    target_link_libraries(scene-history-test PRIVATE mnm-scene-renderer Qt6::Gui)
    target_compile_options(scene-history-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-scene-history COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:scene-history-test>)
    set_tests_properties(native-scene-history PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
