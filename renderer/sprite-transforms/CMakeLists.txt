cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_transforms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET mnm-sprite-renderer)
 add_subdirectory(../sprites "${CMAKE_CURRENT_BINARY_DIR}/sprites")
endif()
add_library(mnm-sprite-transforms STATIC sprite_transforms.cpp)
target_include_directories(mnm-sprite-transforms PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-sprite-transforms PUBLIC mnm-sprite-renderer)
target_compile_options(mnm-sprite-transforms PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
 if(NOT TARGET mnm-sprite-session)
  add_subdirectory(../../apps/sprite-session "${CMAKE_CURRENT_BINARY_DIR}/session")
 endif()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(sprite-transform-test ../../tests/sprite-transform-test.cpp)
 target_link_libraries(sprite-transform-test PRIVATE mnm-sprite-transforms mnm-sprite-session)
 target_compile_options(sprite-transform-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME native-sprite-transforms COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/run-surface-wrapper-control.py" --log-root "${CMAKE_CURRENT_BINARY_DIR}/transform-displays" -- $<TARGET_FILE:sprite-transform-test>)
 set_tests_properties(native-sprite-transforms PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;LP_NUM_THREADS=2" TIMEOUT 150)
endif()
