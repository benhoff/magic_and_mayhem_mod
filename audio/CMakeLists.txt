cmake_minimum_required(VERSION 3.16)
project(mnm_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
if(NOT TARGET mnm-assets)
    add_subdirectory(../assets "${CMAKE_CURRENT_BINARY_DIR}/assets")
endif()
add_library(mnm-audio STATIC buffers.cpp voices.cpp)
target_include_directories(mnm-audio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(mnm-audio PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-audio-reconstruction STATIC ../reconstruction/audio/dsound_setup.cpp
    ../reconstruction/audio/voice_contract.cpp)
target_include_directories(mnm-audio-reconstruction PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../reconstruction/audio")
target_link_libraries(mnm-audio-reconstruction PUBLIC mnm-audio)
target_compile_options(mnm-audio-reconstruction PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-audio-loader STATIC wave_loader.cpp)
target_include_directories(mnm-audio-loader PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-audio-loader PUBLIC mnm-audio mnm-assets)
target_compile_options(mnm-audio-loader PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-audio-upload upload_main.cpp)
target_link_libraries(mnm-audio-upload PRIVATE mnm-audio-reconstruction mnm-audio-loader Qt6::Core)
target_compile_options(mnm-audio-upload PRIVATE -Wall -Wextra -Werror -pedantic)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(audio-buffers-test ../tests/audio-buffers-test.cpp)
    target_link_libraries(audio-buffers-test PRIVATE mnm-audio-reconstruction)
    target_compile_options(audio-buffers-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME audio-buffer-lifecycle COMMAND audio-buffers-test)
    add_executable(audio-voices-test ../tests/audio-voices-test.cpp)
    target_link_libraries(audio-voices-test PRIVATE mnm-audio-reconstruction)
    target_compile_options(audio-voices-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME audio-voice-contracts COMMAND audio-voices-test)
    add_executable(audio-voice-state-test ../tests/audio-voice-state-test.cpp)
    target_link_libraries(audio-voice-state-test PRIVATE mnm-audio-reconstruction)
    target_compile_options(audio-voice-state-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME audio-native-voice-state COMMAND audio-voice-state-test)
    add_test(NAME audio-asset-input COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-audio-asset-input.py" $<TARGET_FILE:mnm-audio-upload>)
endif()
