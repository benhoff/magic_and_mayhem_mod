cmake_minimum_required(VERSION 3.16)
project(mnm_motion_reconstruction LANGUAGES CXX)
add_library(mnm_motion STATIC creature_motion.cpp)
target_include_directories(mnm_motion PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(mnm_motion PUBLIC cxx_std_17)
target_compile_options(mnm_motion PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-motion-test ../../tests/creature-motion-test.cpp)
    target_link_libraries(creature-motion-test PRIVATE mnm_motion)
    target_compile_options(creature-motion-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-motion-arithmetic COMMAND creature-motion-test)
endif()
