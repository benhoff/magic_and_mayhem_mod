cmake_minimum_required(VERSION 3.16)
project(mnm_motion_reconstruction LANGUAGES CXX)
add_library(mnm_motion STATIC creature_motion.cpp segment_setup.cpp)
target_include_directories(mnm_motion PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../pathfinding")
if(NOT TARGET mnm_pathfinding)
    add_subdirectory(../pathfinding pathfinding)
endif()
target_link_libraries(mnm_motion PRIVATE mnm_pathfinding)
target_compile_features(mnm_motion PUBLIC cxx_std_17)
target_compile_options(mnm_motion PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(segment-setup-test ../../tests/segment-setup-test.cpp)
    target_link_libraries(segment-setup-test PRIVATE mnm_motion)
    target_compile_options(segment-setup-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-segment-setup COMMAND segment-setup-test)
    add_executable(creature-motion-test ../../tests/creature-motion-test.cpp)
    target_link_libraries(creature-motion-test PRIVATE mnm_motion)
    target_compile_options(creature-motion-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-motion-arithmetic COMMAND creature-motion-test)
endif()
