cmake_minimum_required(VERSION 3.16)
project(mnm_original_health LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mnm-original-health STATIC health.cpp)
target_include_directories(mnm-original-health PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(mnm-original-health PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-original-melee STATIC melee.cpp)
target_include_directories(mnm-original-melee PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(mnm-original-melee PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(ordinary-health-test ../../tests/ordinary-health-test.cpp)
    target_link_libraries(ordinary-health-test PRIVATE mnm-original-health)
    target_compile_options(ordinary-health-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME ordinary-health-contract COMMAND ordinary-health-test)
endif()
