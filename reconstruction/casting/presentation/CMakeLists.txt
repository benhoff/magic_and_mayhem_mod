cmake_minimum_required(VERSION 3.16)
project(mnm_cure_presentation LANGUAGES CXX)
add_library(mnm-cure-presentation STATIC ../cure_presentation.cpp ../../rendering/effect_animation_binding.cpp ../../rendering/effect_animation_selection.cpp ../../animation/no_cd.cpp)
target_include_directories(mnm-cure-presentation PUBLIC ../../../assets ../../animation)
target_compile_features(mnm-cure-presentation PUBLIC cxx_std_17)
target_compile_options(mnm-cure-presentation PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(cure-presentation-test ../../../tests/cure-presentation-test.cpp)
  target_link_libraries(cure-presentation-test PRIVATE mnm-cure-presentation)
  target_compile_options(cure-presentation-test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME cure-presentation COMMAND cure-presentation-test)
endif()
