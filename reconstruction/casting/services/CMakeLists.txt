cmake_minimum_required(VERSION 3.16)
project(mnm_cure_services LANGUAGES CXX)
add_library(mnm-cure-services STATIC ../cure_services.cpp ../../audio/voice_contract.cpp ../../audio/voice_lifetime.cpp)
target_include_directories(mnm-cure-services PUBLIC ../../../audio)
target_compile_features(mnm-cure-services PUBLIC cxx_std_17)
target_compile_options(mnm-cure-services PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(cure-services-test ../../../tests/cure-services-test.cpp)
  target_link_libraries(cure-services-test PRIVATE mnm-cure-services)
  target_compile_options(cure-services-test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME cure-services COMMAND cure-services-test)
endif()
