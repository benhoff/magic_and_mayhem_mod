cmake_minimum_required(VERSION 3.16)
project(mnm_cast_descriptor LANGUAGES CXX)
add_library(mnm-cast-descriptor STATIC ../descriptor.cpp)
target_compile_features(mnm-cast-descriptor PUBLIC cxx_std_17)
target_compile_options(mnm-cast-descriptor PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(cast-descriptor-test ../../../tests/cast-descriptor-test.cpp)
target_link_libraries(cast-descriptor-test PRIVATE mnm-cast-descriptor)
enable_testing()
add_test(NAME cast-descriptor COMMAND cast-descriptor-test)
