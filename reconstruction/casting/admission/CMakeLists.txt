cmake_minimum_required(VERSION 3.16)
project(mnm_cast_admission LANGUAGES CXX)
add_library(mnm-cast-admission STATIC ../admission.cpp)
target_compile_features(mnm-cast-admission PUBLIC cxx_std_17)
target_compile_options(mnm-cast-admission PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(cast-admission-test ../../../tests/cast-admission-test.cpp)
target_link_libraries(cast-admission-test PRIVATE mnm-cast-admission)
enable_testing()
add_test(NAME cast-admission COMMAND cast-admission-test)
