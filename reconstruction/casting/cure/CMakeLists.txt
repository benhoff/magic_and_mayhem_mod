cmake_minimum_required(VERSION 3.16)
project(mnm_cure_effect LANGUAGES CXX)
add_library(mnm-cure-effect STATIC ../cure_effect.cpp)
target_compile_features(mnm-cure-effect PUBLIC cxx_std_17)
target_compile_options(mnm-cure-effect PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(cure-effect-test ../../../tests/cure-effect-test.cpp)
  target_link_libraries(cure-effect-test PRIVATE mnm-cure-effect)
  target_compile_options(cure-effect-test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME cure-effect COMMAND cure-effect-test)
endif()
