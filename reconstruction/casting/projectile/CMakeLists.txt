cmake_minimum_required(VERSION 3.16)
project(mnm_projectile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mnm-projectile STATIC ../projectile.cpp)
target_compile_options(mnm-projectile PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(projectile-test ../../../tests/projectile-test.cpp)
  target_link_libraries(projectile-test PRIVATE mnm-projectile)
  add_test(NAME projectile-dispatch COMMAND projectile-test)
endif()
