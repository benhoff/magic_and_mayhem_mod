cmake_minimum_required(VERSION 3.16)
project(mnm_original_creature_special_flight LANGUAGES CXX)
if(NOT TARGET mnm-original-creature-routing)
    add_subdirectory(../routing routing)
endif()
add_library(mnm-original-creature-special-flight STATIC ../special_flight_actions.cpp)
target_compile_features(mnm-original-creature-special-flight PUBLIC cxx_std_17)
target_include_directories(mnm-original-creature-special-flight PUBLIC ..)
target_compile_options(mnm-original-creature-special-flight PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(mnm-original-creature-special-flight PUBLIC mnm-original-creature-routing)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-special-flight-test ../../../tests/creature-special-flight-test.cpp)
    target_link_libraries(creature-special-flight-test PRIVATE mnm-original-creature-special-flight)
    target_compile_options(creature-special-flight-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-special-flight-test COMMAND creature-special-flight-test)
endif()
