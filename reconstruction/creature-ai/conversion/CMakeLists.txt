cmake_minimum_required(VERSION 3.16)
project(mnm_creature_owner_conversion LANGUAGES CXX)
add_library(mnm-creature-owner-conversion STATIC ../owner_conversion.cpp)
target_compile_features(mnm-creature-owner-conversion PUBLIC cxx_std_17)

target_link_libraries(creature-owner-conversion-test PRIVATE mnm-creature-owner-conversion)
enable_testing()
add_test(NAME creature-owner-conversion COMMAND creature-owner-conversion-test)
