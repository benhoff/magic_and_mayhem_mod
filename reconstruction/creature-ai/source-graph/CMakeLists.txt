cmake_minimum_required(VERSION 3.16)
project(mnm_creature_source_graph LANGUAGES CXX)
add_library(mnm-creature-source-graph STATIC ../source_graph.cpp)
target_compile_features(mnm-creature-source-graph PUBLIC cxx_std_17)
target_compile_options(mnm-creature-source-graph PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(creature-source-graph-test ../../../tests/creature-source-graph-test.cpp)
target_link_libraries(creature-source-graph-test PRIVATE mnm-creature-source-graph)
enable_testing()
add_test(NAME creature-source-graph COMMAND creature-source-graph-test)
