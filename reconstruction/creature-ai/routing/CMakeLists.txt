cmake_minimum_required(VERSION 3.16)
project(mnm_original_creature_routing LANGUAGES CXX)
if(NOT TARGET mnm-original-creature-segment-setup)
    add_subdirectory(../segment-setup segment-setup)
endif()
add_library(mnm-original-creature-routing STATIC ../actor_displacement.cpp ../actor_route_request.cpp)
target_compile_features(mnm-original-creature-routing PUBLIC cxx_std_17)
target_include_directories(mnm-original-creature-routing PUBLIC ..)
target_compile_options(mnm-original-creature-routing PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(mnm-original-creature-routing PUBLIC mnm-original-creature-segment-setup)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-routing-test ../../../tests/creature-routing-test.cpp)
    target_link_libraries(creature-routing-test PRIVATE mnm-original-creature-routing)
    target_compile_options(creature-routing-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-routing-test COMMAND creature-routing-test)
    add_executable(creature-route-request-test ../../../tests/creature-route-request-test.cpp)
    target_link_libraries(creature-route-request-test PRIVATE mnm-original-creature-routing)
    target_compile_options(creature-route-request-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-route-request-test COMMAND creature-route-request-test)
endif()
