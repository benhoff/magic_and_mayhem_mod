cmake_minimum_required(VERSION 3.16)
project(mnm_contribution_account LANGUAGES CXX)
add_library(mnm-contribution-account STATIC ../contribution_account.cpp)
target_compile_features(mnm-contribution-account PUBLIC cxx_std_17)
target_compile_options(mnm-contribution-account PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(contribution-account-test ../../../tests/contribution-account-test.cpp)
target_link_libraries(contribution-account-test PRIVATE mnm-contribution-account)
enable_testing()
add_test(NAME contribution-account COMMAND contribution-account-test)
