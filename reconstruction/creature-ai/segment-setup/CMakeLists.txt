cmake_minimum_required(VERSION 3.16)
project(mnm_original_creature_segment_setup LANGUAGES CXX)
if(NOT TARGET mnm-original-creature-segments)
    add_subdirectory(../segments segments)
endif()
add_library(mnm-original-creature-segment-setup STATIC ../full_segment_setup.cpp)
target_compile_features(mnm-original-creature-segment-setup PUBLIC cxx_std_17)
target_include_directories(mnm-original-creature-segment-setup PUBLIC ..)
target_compile_options(mnm-original-creature-segment-setup PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(mnm-original-creature-segment-setup PUBLIC mnm-original-creature-segments)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-segment-setup-test ../../../tests/creature-segment-setup-test.cpp)
    target_link_libraries(creature-segment-setup-test PRIVATE mnm-original-creature-segment-setup)
    target_compile_options(creature-segment-setup-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-segment-setup-test COMMAND creature-segment-setup-test)
endif()
