cmake_minimum_required(VERSION 3.16)
project(mnm_creature_resource LANGUAGES CXX)
add_library(mnm-creature-resource STATIC ../resource.cpp)
target_compile_features(mnm-creature-resource PUBLIC cxx_std_17)
target_compile_options(mnm-creature-resource PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(creature-resource-test ../../../tests/creature-resource-test.cpp)
target_link_libraries(creature-resource-test PRIVATE mnm-creature-resource)
enable_testing()
add_test(NAME creature-resource COMMAND creature-resource-test)
