cmake_minimum_required(VERSION 3.16)
project(mnm_original_creature_segments LANGUAGES CXX)
if(NOT TARGET mnm-original-creature-execution)
    add_subdirectory(../execution executor)
endif()
add_library(mnm-original-creature-segments STATIC ../segment_completion.cpp)
target_compile_features(mnm-original-creature-segments PUBLIC cxx_std_17)
target_include_directories(mnm-original-creature-segments PUBLIC ..)
target_compile_options(mnm-original-creature-segments PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(mnm-original-creature-segments PUBLIC mnm-original-creature-execution)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-segment-test ../../../tests/creature-segment-test.cpp)
    target_link_libraries(creature-segment-test PRIVATE mnm-original-creature-segments)
    target_compile_options(creature-segment-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-segment-test COMMAND creature-segment-test)
endif()
