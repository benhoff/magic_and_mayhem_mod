cmake_minimum_required(VERSION 3.16)
project(mnm_creature_ownership_helpers LANGUAGES CXX)
add_library(mnm-creature-ownership-helpers STATIC ../ownership_helpers.cpp)
target_compile_features(mnm-creature-ownership-helpers PUBLIC cxx_std_17)
target_compile_options(mnm-creature-ownership-helpers PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(creature-ownership-helpers-test ../../../tests/creature-ownership-helpers-test.cpp)
target_link_libraries(creature-ownership-helpers-test PRIVATE mnm-creature-ownership-helpers)
enable_testing()
add_test(NAME creature-ownership-helpers COMMAND creature-ownership-helpers-test)
