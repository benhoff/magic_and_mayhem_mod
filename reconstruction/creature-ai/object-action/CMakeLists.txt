cmake_minimum_required(VERSION 3.16)
project(mnm_creature_object_action LANGUAGES CXX)
add_library(mnm-creature-object-action STATIC ../object_action.cpp)
target_compile_features(mnm-creature-object-action PUBLIC cxx_std_17)
target_compile_options(mnm-creature-object-action PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(creature-object-action-test ../../../tests/creature-object-action-test.cpp)
target_link_libraries(creature-object-action-test PRIVATE mnm-creature-object-action)
enable_testing()
add_test(NAME creature-object-action COMMAND creature-object-action-test)
