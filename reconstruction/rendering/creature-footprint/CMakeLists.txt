cmake_minimum_required(VERSION 3.16)
project(mnm_creature_footprint LANGUAGES CXX)
add_library(mnm-creature-footprint STATIC ../creature_footprint.cpp ../creature_occupancy.cpp)
target_compile_features(mnm-creature-footprint PUBLIC cxx_std_17)
target_include_directories(mnm-creature-footprint PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(mnm-creature-footprint PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-footprint-test ../../../tests/creature-footprint-test.cpp)
    target_link_libraries(creature-footprint-test PRIVATE mnm-creature-footprint)
    target_compile_options(creature-footprint-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-footprint-initializer COMMAND creature-footprint-test)
endif()
