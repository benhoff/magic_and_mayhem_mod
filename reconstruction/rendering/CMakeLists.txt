cmake_minimum_required(VERSION 3.16)
project(mnm_sprite_queue LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mnm-sprite-queue STATIC sprite_queue.cpp sprite_visibility.cpp terrain_submission.cpp)
target_include_directories(mnm-sprite-queue PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(mnm-sprite-queue PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(sprite-queue-test ../../tests/sprite-queue-test.cpp)
    target_link_libraries(sprite-queue-test PRIVATE mnm-sprite-queue)
    add_test(NAME sprite-queue-order COMMAND sprite-queue-test)
    add_executable(sprite-visibility-test ../../tests/sprite-visibility-test.cpp)
    target_link_libraries(sprite-visibility-test PRIVATE mnm-sprite-queue)
    add_test(NAME sprite-visibility-mask COMMAND sprite-visibility-test)
endif()
