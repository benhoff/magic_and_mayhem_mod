cmake_minimum_required(VERSION 3.16)
project(mnm_world_dispatch LANGUAGES CXX)
add_library(mnm-world-dispatch INTERFACE)
target_include_directories(mnm-world-dispatch INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(mnm-world-dispatch INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(world-dispatch-test ../../../tests/world-dispatch-test.cpp)
    add_executable(kind8-method-test ../../../tests/kind8-method-test.cpp)
    foreach(target world-dispatch-test kind8-method-test)
        target_link_libraries(${target} PRIVATE mnm-world-dispatch)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
endif()
