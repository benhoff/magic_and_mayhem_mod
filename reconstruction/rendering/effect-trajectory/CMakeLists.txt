cmake_minimum_required(VERSION 3.16)
project(mnm_effect_trajectory LANGUAGES CXX)
add_library(mnm-effect-trajectory STATIC ../effect_trajectory.cpp)
target_compile_features(mnm-effect-trajectory PUBLIC cxx_std_17)
target_include_directories(mnm-effect-trajectory PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(mnm-effect-trajectory PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(effect-trajectory-test ../../../tests/effect-trajectory-test.cpp)
    target_link_libraries(effect-trajectory-test PRIVATE mnm-effect-trajectory)
    target_compile_options(effect-trajectory-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME effect-trajectory-step COMMAND effect-trajectory-test)
endif()
