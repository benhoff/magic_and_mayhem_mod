cmake_minimum_required(VERSION 3.16)
project(mnm_creature_occupancy LANGUAGES CXX)
add_library(mnm-creature-occupancy STATIC ../creature_occupancy.cpp)
target_compile_features(mnm-creature-occupancy PUBLIC cxx_std_17)
target_include_directories(mnm-creature-occupancy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(mnm-creature-occupancy PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(creature-occupancy-test ../../../tests/creature-occupancy-test.cpp)
    target_link_libraries(creature-occupancy-test PRIVATE mnm-creature-occupancy)
    target_compile_options(creature-occupancy-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME creature-occupancy-lookup COMMAND creature-occupancy-test)
endif()
