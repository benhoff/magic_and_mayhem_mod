cmake_minimum_required(VERSION 3.16)
project(mnm_effect_trajectory_initializer LANGUAGES CXX)
add_library(mnm-effect-trajectory-initializer STATIC ../effect_trajectory_initializer.cpp ../effect_trajectory.cpp)
target_compile_features(mnm-effect-trajectory-initializer PUBLIC cxx_std_17)
target_include_directories(mnm-effect-trajectory-initializer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(mnm-effect-trajectory-initializer PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(effect-trajectory-initializer-test ../../../tests/effect-trajectory-initializer-test.cpp)
    target_link_libraries(effect-trajectory-initializer-test PRIVATE mnm-effect-trajectory-initializer)
    target_compile_options(effect-trajectory-initializer-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME effect-trajectory-initializer COMMAND effect-trajectory-initializer-test)
endif()
