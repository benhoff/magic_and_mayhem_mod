cmake_minimum_required(VERSION 3.16)
project(mnm_world_trail LANGUAGES CXX)
add_library(mnm-world-trail INTERFACE)
target_include_directories(mnm-world-trail INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(mnm-world-trail INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(world-trail-test ../../../tests/world-trail-test.cpp)
    target_link_libraries(world-trail-test PRIVATE mnm-world-trail)
    target_compile_options(world-trail-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME world-trail-test COMMAND world-trail-test)
endif()
