cmake_minimum_required(VERSION 3.16)
project(mnm_ground_picking LANGUAGES CXX)
add_library(mnm-ground-picking STATIC ../ground_picking.cpp ../terrain_traversal.cpp ../terrain_submission.cpp ../sprite_queue.cpp)
target_compile_features(mnm-ground-picking PUBLIC cxx_std_17)
target_include_directories(mnm-ground-picking PUBLIC ..)
target_compile_options(mnm-ground-picking PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(ground-picking-test ../../../tests/ground-picking-test.cpp)
    target_link_libraries(ground-picking-test PRIVATE mnm-ground-picking)
    target_compile_options(ground-picking-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME ground-picking-owned-contract COMMAND ground-picking-test)
endif()
