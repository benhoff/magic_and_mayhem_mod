cmake_minimum_required(VERSION 3.16)
project(mnm_effect_creature LANGUAGES CXX)
add_library(mnm-effect-creature STATIC ../effect_transition.cpp ../effect_cleanup.cpp ../effect_projection.cpp ../effect_placement.cpp ../effect_lighting.cpp ../terrain_lighting_cycle.cpp ../terrain_static_lighting.cpp ../terrain_creature_lighting.cpp ../terrain_lighting.cpp ../effect_trajectory.cpp ../creature_occupancy.cpp ../creature_footprint.cpp ../effect_creature_collision.cpp)
target_compile_features(mnm-effect-creature PUBLIC cxx_std_17)
target_include_directories(mnm-effect-creature PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(mnm-effect-creature PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(effect-creature-test ../../../tests/effect-creature-test.cpp)
 target_link_libraries(effect-creature-test PRIVATE mnm-effect-creature)
 target_compile_options(effect-creature-test PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME effect-creature-collision COMMAND effect-creature-test)
endif()
