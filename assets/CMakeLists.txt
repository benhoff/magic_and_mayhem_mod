cmake_minimum_required(VERSION 3.16)
project(mnm_assets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
add_library(mnm-assets STATIC path_resolver.cpp asset_file.cpp)
target_include_directories(mnm-assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-assets PRIVATE Qt6::Core)
target_compile_options(mnm-assets PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-profile-loader STATIC profile.cpp)
target_link_libraries(mnm-profile-loader PUBLIC mnm-assets)
target_compile_options(mnm-profile-loader PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-profile-compare ../tests/profile-api/native.cpp)
target_link_libraries(mnm-profile-compare PRIVATE mnm-profile-loader)
target_compile_options(mnm-profile-compare PRIVATE -Wall -Wextra -Werror -pedantic)

target_link_libraries(mnm-animation-inspect PRIVATE mnm-animation-loader Qt6::Core)
target_compile_options(mnm-animation-inspect PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(mnm-sprite-loader PUBLIC mnm-assets)

add_executable(mnm-asset-compare compare_main.cpp)
