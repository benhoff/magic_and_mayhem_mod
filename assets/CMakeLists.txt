cmake_minimum_required(VERSION 3.16)
project(mnm_assets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
add_library(mnm-assets STATIC path_resolver.cpp asset_file.cpp)
target_include_directories(mnm-assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-assets PRIVATE Qt6::Core)
target_compile_options(mnm-assets PRIVATE -Wall -Wextra -Werror -pedantic)
add_library(mnm-sprite-loader STATIC sprite_loader.cpp)
target_link_libraries(mnm-sprite-loader PUBLIC mnm-assets)
target_compile_options(mnm-sprite-loader PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-sprite-inspect sprite_inspect_main.cpp)
target_link_libraries(mnm-sprite-inspect PRIVATE mnm-sprite-loader Qt6::Core)
target_compile_options(mnm-sprite-inspect PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(mnm-asset-compare compare_main.cpp)
target_link_libraries(mnm-asset-compare PRIVATE mnm-assets Qt6::Core)
target_compile_options(mnm-asset-compare PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(sprite-loader-test ../tests/sprite-loader-test.cpp)
    target_link_libraries(sprite-loader-test PRIVATE mnm-sprite-loader Qt6::Core)
    target_compile_options(sprite-loader-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME sprite-loader COMMAND sprite-loader-test)
    add_executable(asset-path-test ../tests/asset-path-test.cpp)
    target_link_libraries(asset-path-test PRIVATE mnm-assets Qt6::Core)
    target_compile_options(asset-path-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME asset-path-resolution COMMAND asset-path-test)
    add_executable(asset-file-test ../tests/asset-file-test.cpp)
    target_link_libraries(asset-file-test PRIVATE mnm-assets Qt6::Core)
    target_compile_options(asset-file-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME asset-file-io COMMAND asset-file-test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME asset-byte-comparison COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test-asset-compare.py" $<TARGET_FILE:mnm-asset-compare>)
endif()
