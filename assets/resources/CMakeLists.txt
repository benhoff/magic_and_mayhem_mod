cmake_minimum_required(VERSION 3.16)
project(mnm_resources LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui)
if(NOT TARGET mnm-assets)
    add_subdirectory(.. "${CMAKE_CURRENT_BINARY_DIR}/assets")
endif()
add_library(mnm-resources STATIC resource_manager.cpp)
target_include_directories(mnm-resources PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mnm-resources PUBLIC mnm-sprite-loader mnm-animation-loader mnm-terrain-catalog mnm-bmp-loader mnm-pcx-loader mnm-jpeg-loader)
target_compile_options(mnm-resources PRIVATE -Wall -Wextra -Werror -pedantic)
include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(resource-manager-test ../../tests/resource-manager-test.cpp)
    target_link_libraries(resource-manager-test PRIVATE mnm-resources Qt6::Core Qt6::Gui Threads::Threads)
    target_compile_options(resource-manager-test PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME native-resource-manager COMMAND resource-manager-test)
endif()
